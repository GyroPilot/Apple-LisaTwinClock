unit TwcNet;

{ REV D  17-Oct-2026  LisaTwinClock 1.1: the time from the network.     }
{                      New unit.  NOT yet compiled on a Lisa.            }

{ TWC-NET.TEXT -- where LisaTwinClock gets the time from the network,   }
{ through an RS-232 FujiNet on Serial B (LisaFuji's FnBus unit, linked  }
{ as it is).  Three sources, chosen on the Network menu and kept in     }
{ TWCNET.DATA beside the tool, with the speed:                          }
{   1  the FujiNet's own clock (network time, in the time zone set on   }
{      the FujiNet: its web page or fnconfig.ini);                       }
{   2  timeapi.io for your town: LisaFuji's HOMETOWN.DATA (T in         }
{      LisaFuji) gives the latitude and longitude, and timeapi.io the   }
{      local time there, summer time included;                           }
{   3  timeapi.io for a time zone you type, such as America/New_York.   }
{ timeapi.io answers one line of JSON: "year":2026,"month":10,"day":17, }
{ "hour":20,"minute":40,"seconds":42 and more; the pieces between the   }
{ commas are read as they arrive, so no buffer is needed.               }
{ TWCNET.DATA is one line:  TN1|source|baud|zone <CR>                   }

INTERFACE

uses (*$U UnitStd.obj *)   UnitStd,
     (*$U UnitHz.obj *)    UnitHz,
     (*$U Storage.obj *)   Storage,
     (*$U QuickDraw.obj *) QuickDraw,
     (*$U SysCall.obj *)   SysCall,
     (*$U TWC/FNBUS.OBJ *) FnBus;

const
   ntSFuji  = 1;              { the sources }
   ntSTown  = 2;
   ntSZone  = 3;
   ntBufLen = 128;            { bytes of a note file }
   ntZCap   = 40;             { characters of a time zone name }
   ntIdle   = 15000;          { ms a fetch waits with nothing arriving }

type
   NtByte   = -128..127;
   NtZStr   = string[40];

var
   ntSrc    : integer;        { the source in use }
   ntBaud   : longint;        { 9600 or 19200 }
   ntZone   : NtZStr;         { for source 3 }
   ntTown   : string[60];     { from HOMETOWN.DATA }
   ntLat    : longint;        { degrees times 10,000 }
   ntLon    : longint;
   ntHave   : boolean;        { a town was read }
   ntWhy    : string[80];     { why NtGet failed, in words }
   ntRc     : integer;        { FnBus result of a failed call }
   ntEcode  : integer;        { OS ecode of the last note-file call }
   ntBuf    : array [0..ntBufLen] of NtByte;

   { a reply being read }
   ntGot    : integer;        { 1 year 2 month 4 day 8 hour 16 minute }
   ntY, ntMo, ntD, ntH, ntMi, ntS : integer;

procedure NtInit;
function  NtLoad   (var vol : E_Name) : boolean;
function  NtSave   (var vol : E_Name) : boolean;
function  NtTownLd (var vol : E_Name) : boolean;
function  NtGet    (var y, mo, d, h, mi, s : integer) : integer;


IMPLEMENTATION

{$S TWCNet}

procedure NtInit;
begin
   ntSrc  := ntSFuji;
   ntBaud := 9600;
   ntZone := '';
   ntTown := '';
   ntLat  := 0;
   ntLon  := 0;
   ntHave := false;
   ntWhy  := '';
   ntRc   := 0;
   ntEcode := 0;
   FnInit
end;

function NtVal (var v : Str255) : longint;
{ The whole number at the start of v (a minus sign allowed). }
var i   : integer;
    n   : longint;
    neg : boolean;
    go  : boolean;
begin
   n := 0;
   i := 1;
   neg := false;
   if length(v) > 0 then
      if v[1] = '-' then begin
         neg := true;
         i := 2
      end;
   go := true;
   while go and (i <= length(v)) do begin
      if (v[i] >= '0') and (v[i] <= '9') then begin
         if n < 20000000 then
            n := n * 10 + (ord(v[i]) - 48)
      end
      else
         go := false;
      i := i + 1
   end;
   if neg then n := -n;
   NtVal := n
end;

procedure NtNum (n : longint; var txt : Str255);
{ Decimal text of n. }
var one : string[1];
    neg : boolean;
begin
   neg := n < 0;
   if neg then n := -n;
   txt := '';
   one := ' ';
   repeat
      one[1] := chr(48 + (n mod 10));
      txt := concat(one, txt);
      n := n div 10
   until n = 0;
   if neg then txt := concat('-', txt)
end;

procedure NtDeg (v : longint; var txt : Str255);
{ Degrees times 10,000 as 39.1075 or -84.4958. }
var w, f : Str255;
    neg  : boolean;
begin
   neg := v < 0;
   if neg then v := -v;
   NtNum(v div 10000, w);
   NtNum(v mod 10000 + 10000, f);
   delete(f, 1, 1);
   txt := concat(w, concat('.', f));
   if neg then txt := concat('-', txt)
end;

{------------------------------------------------------------------------}
{ Note files.  A short read comes with a nonzero ecode, so the byte     }
{ count decides (the LISACom lesson).                                    }
{------------------------------------------------------------------------}

function NtRead (var vol : E_Name; nm : Str255; var f0, f1, f2, f3 : Str255) : boolean;
{ A note file of four fields split by bars, into f0 to f3. }
var path     : Pathname;
    refnum   : integer;
    got      : longint;
    ec       : integer;
    schedErr : integer;
    i, b     : integer;
    fld      : integer;
    one      : string[1];
begin
   NtRead := false;
   f0 := '';
   f1 := '';
   f2 := '';
   f3 := '';
   path := concat(vol, nm);
   Sched_Class(schedErr, false);
   OPEN(ec, path, refnum, [DRead]);
   Sched_Class(schedErr, true);
   if ec > 0 then begin
      ntEcode := ec;
      exit(NtRead)
   end;
   got := 0;
   Sched_Class(schedErr, false);
   READ_DATA(ec, refnum, ord4(@ntBuf), ntBufLen, got, sequential, 0);
   CLOSE_OBJECT(ec, refnum);
   Sched_Class(schedErr, true);
   if got < 4 then
      exit(NtRead);
   fld := 0;
   one := ' ';
   for i := 0 to got - 1 do begin
      b := ntBuf[i];
      if b < 0 then b := b + 256;
      if (b = 124) and (fld < 3) then
         fld := fld + 1
      else if (b >= 32) and (b <= 126) then begin
         one[1] := chr(b);
         if fld = 0 then begin
            if length(f0) < 8 then f0 := concat(f0, one)
         end
         else if fld = 1 then begin
            if length(f1) < 12 then f1 := concat(f1, one)
         end
         else if fld = 2 then begin
            if length(f2) < 12 then f2 := concat(f2, one)
         end
         else if length(f3) < 60 then
            f3 := concat(f3, one)
      end
   end;
   NtRead := fld = 3
end;

function NtLoad {var vol : E_Name} : boolean;
{ The source, speed and zone chosen last time. }
var f0, f1, f2, f3 : Str255;
    n              : longint;
begin
   NtLoad := false;
   if not NtRead(vol, 'TWCNET.DATA', f0, f1, f2, f3) then
      exit(NtLoad);
   if f0 <> 'TN1' then
      exit(NtLoad);
   n := NtVal(f1);
   if (n >= ntSFuji) and (n <= ntSZone) then
      ntSrc := n;
   n := NtVal(f2);
   if (n = 9600) or (n = 19200) then
      ntBaud := n;
   if length(f3) > ntZCap then
      f3[0] := chr(ntZCap);
   ntZone := f3;
   NtLoad := true
end;

function NtSave {var vol : E_Name} : boolean;
{ TN1|source|baud|zone <CR>.  890 from MAKE_FILE = it already exists,   }
{ which is fine (TwSave's way).  FALSE and ntEcode on failure.          }
var path     : Pathname;
    refnum   : integer;
    ec       : integer;
    schedErr : integer;
    wrote    : longint;
    line     : Str255;
    part     : Str255;
    i, n     : integer;
begin
   NtSave := false;
   NtNum(ntSrc, part);
   line := concat('TN1|', concat(part, '|'));
   NtNum(ntBaud, part);
   line := concat(line, concat(part, concat('|', ntZone)));
   n := length(line);
   for i := 1 to n do
      ntBuf[i - 1] := ord(line[i]);
   ntBuf[n] := 13;
   n := n + 1;
   path := concat(vol, 'TWCNET.DATA');
   Sched_Class(schedErr, false);
   MAKE_FILE(ntEcode, path, 0);
   Sched_Class(schedErr, true);
   if (ntEcode > 0) and (ntEcode <> 890) then
      exit(NtSave);
   Sched_Class(schedErr, false);
   OPEN(ntEcode, path, refnum, [DWrite]);
   Sched_Class(schedErr, true);
   if ntEcode > 0 then
      exit(NtSave);
   Sched_Class(schedErr, false);
   WRITE_DATA(ntEcode, refnum, ord4(@ntBuf), n, wrote, sequential, 0);
   if (ntEcode <= 0) and (wrote = n) then begin
      TRUNCATE(ec, refnum);
      NtSave := true
   end;
   CLOSE_OBJECT(ec, refnum);
   Sched_Class(schedErr, true);
   if ntEcode < 0 then ntEcode := 0
end;

function NtTownLd {var vol : E_Name} : boolean;
{ Your town, from LisaFuji's HOMETOWN.DATA:  HT1|lat|lon|name.  The    }
{ file must be on the same disk as this tool.                           }
var f0, f1, f2, f3 : Str255;
    la, lo         : longint;
begin
   NtTownLd := false;
   ntHave := false;
   if not NtRead(vol, 'HOMETOWN.DATA', f0, f1, f2, f3) then
      exit(NtTownLd);
   if (f0 <> 'HT1') or (length(f1) = 0) or (length(f2) = 0) then
      exit(NtTownLd);
   la := NtVal(f1);
   lo := NtVal(f2);
   if (la < -900000) or (la > 900000) or (lo < -1800000) or (lo > 1800000) then
      exit(NtTownLd);
   if length(f3) = 0 then
      f3 := 'your town';
   ntTown := f3;
   ntLat := la;
   ntLon := lo;
   ntHave := true;
   NtTownLd := true
end;

{------------------------------------------------------------------------}
{ The network.                                                           }
{------------------------------------------------------------------------}

procedure NtPiece (var s : Str255);
{ One piece of the reply between commas, such as "hour":20 (the first  }
{ piece starts with the open brace).  The key is between the last two   }
{ quotes before the colon.                                               }
var p, q, i : integer;
    key     : Str255;
    val     : Str255;
    n       : integer;
    go      : boolean;
begin
   p := 0;
   for i := 1 to length(s) - 1 do
      if (s[i] = '"') and (s[i + 1] = ':') then
         p := i;
   if p < 2 then
      exit(NtPiece);
   q := p - 1;
   while (q > 0) and (s[q] <> '"') do
      q := q - 1;
   if q = 0 then
      exit(NtPiece);
   key := copy(s, q + 1, p - q - 1);
   val := '';
   if p + 2 <= length(s) then
      val := copy(s, p + 2, length(s) - p - 1);
   go := true;
   while go and (length(val) > 0) do
      if (val[1] = ' ') or (val[1] = '"') then
         delete(val, 1, 1)
      else
         go := false;
   n := NtVal(val);
   if key = 'year' then begin
      ntY := n;
      ntGot := ntGot + 1
   end
   else if key = 'month' then begin
      ntMo := n;
      ntGot := ntGot + 2
   end
   else if key = 'day' then begin
      ntD := n;
      ntGot := ntGot + 4
   end
   else if key = 'hour' then begin
      ntH := n;
      ntGot := ntGot + 8
   end
   else if key = 'minute' then begin
      ntMi := n;
      ntGot := ntGot + 16
   end
   else if key = 'seconds' then
      ntS := n
end;

function NtWeb (var url : Str255) : boolean;
{ Fetch url through the FujiNet and read the time out of it.  FALSE:    }
{ ntWhy says why.  The read loop is LisaFuji's NwGet, without the pane. }
var rc, k, i, b : integer;
    idle        : longint;
    t0          : longint;
    clk         : boolean;
    done, fail  : boolean;
    cur         : Str255;
    one         : string[1];
    part        : Str255;
begin
   NtWeb := false;
   ntGot := 0;
   ntS := 0;
   rc := FnNClos;                       { a channel left open by a failed fetch }
   rc := FnNOpen(url, 4, 0);
   if rc <> fnOK then begin
      ntRc := rc;
      ntWhy := 'the FujiNet did not take the request';
      rc := FnNClos;
      exit(NtWeb)
   end;
   if fnStat <> 1 then begin
      NtNum(fnStat, part);
      if fnStat = 136 then
         ntWhy := 'timeapi.io sent nothing back'
      else if fnStat = 170 then
         ntWhy := 'timeapi.io was not found (status 170): is WiFi up?'
      else if fnStat = 207 then
         ntWhy := 'no connection to timeapi.io (status 207)'
      else
         ntWhy := concat('timeapi.io refused it, status ', part);
      rc := FnNClos;
      exit(NtWeb)
   end;
   idle := 0;
   FnTStart(t0, clk);
   done := false;
   fail := false;
   cur := '';
   one := ' ';
   while not (done or fail) do begin
      if fnWait = 0 then begin
         rc := FnNStat;
         if rc <> fnOK then
            fail := true
         else if fnWait = 0 then begin
            if fnStat = 136 then
               done := true
            else if (fnConn = 0) and (fnStat <> 1) then
               done := true
            else begin
               FnNap(100);
               idle := idle + 100;
               if FnTOver(t0, clk, idle, ntIdle) then begin
                  fail := true;
                  rc := fnNoRep
               end
            end
         end
      end
      else begin
         k := fnRdMax;
         if fnWait < k then
            k := fnWait;
         rc := FnNRead(k);
         if rc <> fnOK then
            fail := true
         else begin
            idle := 0;
            FnTStart(t0, clk);
            for i := 0 to fnPlLen - 1 do begin
               b := FnPlB(i);
               if (b = 44) or (b = 125) or (b = 10) then begin
                  NtPiece(cur);
                  cur := ''
               end
               else if (b >= 32) and (b < 127) and (length(cur) < 120) then begin
                  one[1] := chr(b);
                  cur := concat(cur, one)
               end
            end
         end
      end
   end;
   if length(cur) > 0 then
      NtPiece(cur);
   k := FnNClos;
   if fail then begin
      ntRc := rc;
      ntWhy := 'the reply from timeapi.io stopped part-way';
      exit(NtWeb)
   end;
   if ntGot <> 31 then begin
      ntWhy := 'timeapi.io answered, but with no time in it (a wrong zone name?)';
      exit(NtWeb)
   end;
   NtWeb := true
end;

function NtGet {var y, mo, d, h, mi, s : integer} : integer;
{ The time from the source in use.  0 = it is in y .. s.               }
{ 1 = Serial B would not open; 2 = the FujiNet or the web failed (ntWhy, }
{ ntRc); 3 = no town; 4 = no time zone.                                 }
var url  : Str255;
    a, b : Str255;
    rc   : integer;
    ok   : boolean;
begin
   y := 0;
   mo := 0;
   d := 0;
   h := 0;
   mi := 0;
   s := 0;
   ntRc := 0;
   ntWhy := '';
   if (ntSrc = ntSTown) and (not ntHave) then begin
      NtGet := 3;
      exit(NtGet)
   end;
   if (ntSrc = ntSZone) and (length(ntZone) = 0) then begin
      NtGet := 4;
      exit(NtGet)
   end;
   if not FnPortUp(ntBaud) then begin
      NtGet := 1;
      exit(NtGet)
   end;
   if ntSrc = ntSFuji then begin
      rc := FnClock(d, mo, y, h, mi, s);
      ok := rc = fnOK;
      if not ok then begin
         ntRc := rc;
         ntWhy := 'the FujiNet did not give the time'
      end
   end
   else begin
      if ntSrc = ntSTown then begin
         NtDeg(ntLat, a);
         NtDeg(ntLon, b);
         url := concat('N1:HTTPS://timeapi.io/api/time/current/coordinate?latitude=', a);
         url := concat(url, concat('&longitude=', b))
      end
      else
         url := concat('N1:HTTPS://timeapi.io/api/time/current/zone?timeZone=', ntZone);
      ok := NtWeb(url);
      if ok then begin
         y := ntY;
         mo := ntMo;
         d := ntD;
         h := ntH;
         mi := ntMi;
         s := ntS
      end
   end;
   FnPortDn;
   if ok then
      NtGet := 0
   else
      NtGet := 2
end;
